cmake_minimum_required(VERSION 3.22)
if(NOT DEFINED SDKCONFIG)
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
endif()
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(splitflap_s3_bringup VERSION 0.3.0)
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" VERSION_EQUAL "5.5.6")
    message(FATAL_ERROR "Use the validated ESP-IDF v5.5.6 release.")
endif()
if(NOT IDF_TARGET STREQUAL "esp32s3")
    message(FATAL_ERROR "Standalone bring-up is for esp32s3; the existing ESP32 app remains in ../esp-idf.")
endif()
if(CONFIG_SPIRAM)
    message(FATAL_ERROR "Unknown-board bring-up keeps PSRAM disabled; do not select a bus mode before identifying the memory.")
endif()
if(NOT CONFIG_ESP_CONSOLE_UART_DEFAULT OR NOT CONFIG_ESP_CONSOLE_UART_NUM EQUAL 0)
    message(FATAL_ERROR "Bring-up requires default UART0; reported Chainlink pins must never become console outputs.")
endif()
